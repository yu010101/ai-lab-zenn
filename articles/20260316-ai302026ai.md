### ROI（投資対効果）の計算

```python
def calculate_ai_tool_roi(monthly_salary: int, productivity_gain: float, 
                          tool_cost: int, months: int = 12) -> dict:
    """
    AIツール導入のROI計算
    
    Args:
        monthly_salary: 月給（円）
        productivity_gain: 生産性向上率（0.3 = 30%）
        tool_cost: ツール年間コスト（円）
        months: 計算期間（月）
    
    Returns:
        ROI分析結果
    """
    annual_salary = monthly_salary * months
    productivity_value = annual_salary * productivity_gain
    net_benefit = productivity_value - tool_cost
    roi_percentage = (net_benefit / tool_cost) * 100
    
    return {
        "productivity_value": productivity_value,
        "tool_cost": tool_cost,
        "net_benefit": net_benefit,
        "roi_percentage": round(roi_percentage, 2),
        "payback_months": round(tool_cost / (productivity_value / months), 1)
    }

# 使用例
result = calculate_ai_tool_roi(
    monthly_salary=600000,  # 月給60万円
    productivity_gain=0.30,  # 30%向上
    tool_cost=120000        # 年間12万円
)

print(f"ROI: {result['roi_percentage']}%")
print(f"回収期間: {result['payback_months']}ヶ月")
```

```python
import matplotlib.pyplot as plt
import numpy as np

def create_skill_priority_matrix():
    """
    技術スキルの需要度と習得難易度のマトリクス
    """
    skills = {
        'Python/AI': (9, 6),
        'LLM活用': (8, 4),
        'MLOps': (7, 8),
        'クラウドAI': (8, 5),
        'データ分析': (7, 6),
        'DevOps': (6, 7),
        'セキュリティ': (8, 7),
        'フロントエンド': (5, 5)
    }
    
    # 優先度計算：(需要度 * 重要度係数) - (習得コスト)
    priorities = {}
    for skill, (demand, difficulty) in skills.items():
        priority_score = (demand * 1.5) - difficulty
        priorities[skill] = priority_score
    
    # 降順でソート
    sorted_priorities = sorted(priorities.items(), 
                             key=lambda x: x[1], reverse=True)
    
    return sorted_priorities

# 実行結果
print("2026年技術習得優先順位:")
for i, (skill, score) in enumerate(create_skill_priority_matrix(), 1):
    print(f"{i}. {skill}: {score:.1f}ポイント")
```

```python
            }
        }
    
    def recommend_specialization(self, interests: List[str], 
                               market_demand: Dict[str, float]) -> str:
        # 個人の興味と市場需要を考慮した推奨
        scores = {}
        for spec, data in self.specializations.items():
            interest_score = 1 if spec.lower() in [i.lower() for i in interests] else 0
            demand_score = market_demand.get(spec, 0)
            scores[spec] = (interest_score * 0.6) + (demand_score * 0.4)
        
        return max(scores, key=scores.get)
```
